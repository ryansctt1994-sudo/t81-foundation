124inline T81Symbol T81Symbol::intern(std::string_view sv) noexcept {
125 struct InternTable {
126 std::mutex mtx;
127 std::unordered_map<std::string, id_t> map;
128 id_t next_id = 5; // 0-4 reserved
129 };
130 static InternTable table;
131
132 // Normalize
133 while (!sv.empty() && std::isspace(sv.front())) sv.remove_prefix(1);
134 while (!sv.empty() && std::isspace(sv.back())) sv.remove_suffix(1);
135
136 std::string s(sv); // needed for map key
137
138 std::lock_guard<std::mutex> lock(table.mtx);
139
140 auto it = table.map.find(s);
141 if (it != table.map.end()) {
142 return T81Symbol::from_id(it->second);
143 }
144
145 id_t id = table.next_id++;
146 table.map[s] = id;
147 return T81Symbol::from_id(id);
148}
