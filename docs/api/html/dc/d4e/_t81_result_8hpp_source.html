28// ======================================================================
29// T81Error – A failure that remembers why, when, and how much it cost
30// ======================================================================
31struct T81Error {
32 T81Symbol code; // e.g. symbols::DIVISION_BY_ZERO
33 T81String message;
34 T81Time occurred_at;
35 T81Entropy fuel_spent; // how much was wasted trying
36 T81Symbol source; // which agent/function caused it
37
38 T81Error(T81Symbol c, T81String m, T81Symbol src = symbols::unk)
39 : code(c)
40 , message(std::move(m))
41 , occurred_at(T81Time::now(src))
42 , fuel_spent(acquire_entropy())
43 , source(src)
44 {}
45
46 T81Error(const T81Error& other)
47 : code(other.code)
48 , message(other.message)
49 , occurred_at(T81Time::now(T81Symbol::intern("ERROR_COPY")))
50 , fuel_spent(acquire_entropy()) // copying an error costs fresh entropy
51 , source(other.source)
52 {}
53
54 [[nodiscard]] T81String explain() const {
55 return T81String("[ERROR ") + T81String(code.to_string())
56 + T81String(" at ") + occurred_at.narrate()
57 + T81String("] ") + message
58 + T81String(" (source: ") + T81String(source.to_string()) + T81String(")");
59 }
60};
