23inline int hex_digit(char value) {
24 if (value >= '0' && value <= '9') return value - '0';
25 if (value >= 'a' && value <= 'f') return 10 + (value - 'a');
26 if (value >= 'A' && value <= 'F') return 10 + (value - 'A');
27 return -1;
28}

30inline std::string decode_string_literal(const Token& token) {
31 std::string_view view = token.lexeme;
32 if (view.size() < 2 || view.front() != '"' || view.back() != '"') {
33 return {};
34 }
35 std::string result;
36 result.reserve(view.size() - 2);
37 for (size_t i = 1; i + 1 < view.size(); ++i) {
38 char c = view[i];
39 if (c == '\\' && i + 1 < view.size() - 1) {
40 ++i;
41 char esc = view[i];
42 switch (esc) {
43 case '\\': result.push_back('\\'); break;
44 case '"': result.push_back('"'); break;
45 case 'n': result.push_back('\n'); break;
46 case 'r': result.push_back('\r'); break;
47 case 't': result.push_back('\t'); break;
48 case 'x': {
49 if (i + 2 < view.size() - 1) {
50 int hi = hex_digit(view[++i]);
51 int lo = hex_digit(view[++i]);
52 if (hi >= 0 && lo >= 0) {
53 result.push_back(static_cast<char>((hi << 4) | lo));
54 }
55 }
56 break;
57 }
58 default: result.push_back(esc); break;
59 }
60 } else {
61 result.push_back(c);
62 }
63 }
64 return result;
65}

308 NumericCategory left_cat = categorize(left_type);
309 NumericCategory right_cat = categorize(right_type);
310 NumericCategory target_category = left_cat;
311 auto merge_category = [](NumericCategory a, NumericCategory b) {
312 if (a == NumericCategory::Float || b == NumericCategory::Float) return NumericCategory::Float;
313 if (a == NumericCategory::Fraction || b == NumericCategory::Fraction) return NumericCategory::Fraction;
314 if (a == NumericCategory::Integer || b == NumericCategory::Integer) return NumericCategory::Integer;
315 return NumericCategory::Unknown;
316 };
317 target_category = merge_category(left_cat, right_cat);
318 if (target_category == NumericCategory::Unknown) target_category = merge_category(target_category, right_cat);
319
320 tisc::ir::PrimitiveKind operand_primitive = primitive_kind;
321 if (!both_bool) {
322 switch (target_category) {
323 case NumericCategory::Float:
324 operand_primitive = tisc::ir::PrimitiveKind::Float;
325 break;
326 case NumericCategory::Fraction:
327 operand_primitive = tisc::ir::PrimitiveKind::Fraction;
328 break;
329 case NumericCategory::Integer:
330 operand_primitive = tisc::ir::PrimitiveKind::Integer;
331 break;
332 default:
333 operand_primitive = left.primitive;
334 break;
335 }
336 } else {
337 operand_primitive = tisc::ir::PrimitiveKind::Integer;
338 }
339

346 instr.boolean_result = true;
347 instr.relation = relation;
348 emit(instr);
349 record_result(&expr, dest);
350 return {};
351 }

354 throw std::runtime_error("Modulo requires integer operands");
355 }
356
357 auto left_converted = ensure_kind(left, primitive_kind);
358 auto right_converted = ensure_kind(right, primitive_kind);
359 auto dest = allocate_typed_register(primitive_kind);
360 using O = tisc::ir::Opcode;
361 tisc::ir::Opcode opcode;
362 switch (expr.op.type) {
363 case TokenType::Plus:
364 opcode = select_opcode(kind, O::ADD, O::FADD, O::FRACADD);
365 break;
366 case TokenType::Minus:
367 opcode = select_opcode(kind, O::SUB, O::FSUB, O::FRACSUB);
368 break;
369 case TokenType::Star:
370 opcode = select_opcode(kind, O::MUL, O::FMUL, O::FRACMUL);
371 break;
372 case TokenType::Slash:
373 opcode = select_opcode(kind, O::DIV, O::FDIV, O::FRACDIV);
374 break;
375 case TokenType::Percent:
376 opcode = O::MOD;
377 break;
378 default:
379 throw std::runtime_error("Unsupported binary operator");
380 }
381
382 auto instr = tisc::ir::Instruction{opcode, {dest.reg, left_converted.reg, right_converted.reg}};
383 instr.primitive = primitive_kind;
384 emit(instr);
385 record_result(&expr, dest);
386 return {};
387 }
