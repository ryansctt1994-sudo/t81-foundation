85 // Compute strides
86 auto strides = [](const std::vector<int>& s) {
87 std::vector<std::size_t> st(s.size(), 1);
88 for (int i = (int)s.size() - 2; i >= 0; --i) {
89 std::size_t next = st[(std::size_t)(i + 1)];
90 std::size_t dim = static_cast<std::size_t>(s[(std::size_t)(i + 1)]);
91 if (next != 0 && dim > std::numeric_limits<std::size_t>::max() / next) {
92 throw std::overflow_error("broadcast: stride overflow");
93 }
