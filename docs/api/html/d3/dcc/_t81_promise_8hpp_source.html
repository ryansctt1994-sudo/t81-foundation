86 // The sacred act of waiting
87 //===================================================================
88 [[nodiscard]] T81Result<T> await(T81Entropy patience, T81Agent& dreamer) {
89 if (!coro_) {
90 return T81Result<T>::failure(symbols::PROMISE_DESTROYED,
91 "Promise was destroyed before resolution"_t81);
92 }
93
94 if (coro_.done()) {
95 if (coro_.promise().value) {
96 return T81Result<T>::success(std::move(*coro_.promise().value));
97 }
98 if (coro_.promise().error) {
99 return T81Result<T>::failure(coro_.promise().error->code,
100 coro_.promise().error->message);
101 }
102 }
103
104 // Still waiting — pay entropy and dream
105 consume_entropy(patience);
106 dreamer.observe(symbols::DREAMING);
107 record_event(T81Time::now(patience, symbols::WAITING));
108
109 return T81Result<T>::failure(symbols::STILL_DREAMING,
110 "Computation not yet complete — patience is a virtue"_t81);
111 }
