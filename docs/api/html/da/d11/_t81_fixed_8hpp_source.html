29template <std::size_t IntegerTrits, std::size_t FractionalTrits>
30class T81Fixed {
31 static_assert(IntegerTrits > 0, "T81Fixed: IntegerTrits must be > 0");
32 static_assert(FractionalTrits > 0, "T81Fixed: FractionalTrits must be > 0");
33 static_assert(IntegerTrits + FractionalTrits <= 162,
34 "T81Fixed: total trits too large for current kernels");
35
36public:
37 using Storage = T81Int<IntegerTrits + FractionalTrits>;
38
39 static constexpr std::size_t I = IntegerTrits;
40 static constexpr std::size_t F = FractionalTrits;
41 static constexpr std::size_t Total = I + F;
42
43private:
44 Storage data_{};
45
46 [[nodiscard]] static double frac_scale() noexcept {
47 static const double kScale =
48 std::pow(3.0, static_cast<int>(FractionalTrits));
49 return kScale;
50 }
51
52public:
53 // ------------------------------------------------------------------
54 // Construction
55 // ------------------------------------------------------------------
56
57 constexpr T81Fixed() noexcept = default;
58
59 explicit constexpr T81Fixed(const Storage& v) noexcept
60 : data_(v) {}
61
62 explicit constexpr T81Fixed(Storage&& v) noexcept
63 : data_(std::move(v)) {}
64
65 // From signed 64-bit integer (placed into the integer part)
66 explicit constexpr T81Fixed(std::int64_t v) noexcept
67 : data_(Storage(v) << F) {}
68
69 // From double (scaled by 3^F and rounded to nearest)
70 explicit T81Fixed(double v) {
71 const double scaled = v * frac_scale();
72 const std::int64_t rounded = static_cast<std::int64_t>(std::llround(scaled));
73 data_ = Storage(rounded);
74 }
75
76 // Convenience factory
77 [[nodiscard]] static T81Fixed from_double(double v) {
78 return T81Fixed(v);
79 }
80
81 // ------------------------------------------------------------------
82 // Conversion back to double
83 // ------------------------------------------------------------------
84
85 [[nodiscard]] double to_double() const {
86 const std::int64_t raw = data_.to_int64(); // may throw on overflow
87 return static_cast<double>(raw) / frac_scale();
88 }

102 [[nodiscard]] T81Fixed operator*(const T81Fixed& o) const noexcept {
103 // (a * b) has 2F fractional trits; shift right by F to renormalize.
104 return T81Fixed(Storage((raw() * o.raw()) >> F));
105 }
106
107 [[nodiscard]] T81Fixed operator/(const T81Fixed& o) const {
108 // Up-scale numerator first to preserve F fractional trits.
109 if (o.raw().is_zero()) {
110 throw std::domain_error("T81Fixed: division by zero");
111 }
112 return T81Fixed(Storage((raw() << F) / o.raw()));
113 }
