27// ======================================================================
28// Core concept – any scalar that is exactly one tryte (81 trits)
29// ======================================================================
30template <typename T>
31concept T81TryteScalar =
32 std::same_as<T, T81Int<81>> || // exact integer symbols / IDs
33 std::same_as<T, T81Float<72,9>> || // 81-trit float (72+9+sign)
34 std::same_as<T, T81Fixed<72,9>> || // 81-trit fixed Q72.9
35 std::same_as<T, T81Complex<40>>; // 80 trits → rounded to 81 with padding

38// T81Matrix<Scalar, Rows, Cols> – the final matrix type
39// ======================================================================
40template <typename Scalar, size_t Rows, size_t Cols>
41 requires T81TryteScalar<Scalar>
42class T81Matrix {

106 // Matrix × Matrix — the crown jewel (single tensor-core instruction on real silicon)
107 template <size_t K>
108 [[nodiscard]] friend constexpr auto operator*(
109 const T81Matrix<Scalar, Rows, K>& A,
110 const T81Matrix<Scalar, K, Cols>& B) noexcept
111 -> T81Matrix<Scalar, Rows, Cols>
112 {
113 T81Matrix<Scalar, Rows, Cols> C(Scalar(0));
114 for (size_t i = 0; i < Rows; ++i)
115 for (size_t j = 0; j < Cols; ++j) {
116 Scalar sum{};
117 for (size_t k = 0; k < K; ++k)
118 sum = sum + A(i,k) * B(k,j);
119 C(i,j) = sum;
120 }
121 return C;
122 }
