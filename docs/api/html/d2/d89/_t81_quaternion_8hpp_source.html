22// ======================================================================
23// T81Quaternion – Exact quaternion using two T81Complex<27>
24// ======================================================================
25//
26// Representation:
27// We encode the quaternion components (w, x, y, z) as two complex numbers:
28// real_imag = w + x·i
29// j_k = y + z·i
30//
31// This keeps the storage compact and naturally aligned with the
32// existing balanced-ternary complex type.
33// ======================================================================
34class T81Quaternion {
35public:
36 using Scalar = T81Float<27, 9>;
37 using Complex = T81Complex<27>;

39private:

66 // From axis-angle (axis is expected to be normalized).
67 // Exact when angle = k·π/3^n in the underlying ternary representation.
68 static inline T81Quaternion from_axis_angle(
69 const Scalar& axis_x,
70 const Scalar& axis_y,
71 const Scalar& axis_z,
72 const Scalar& angle_radians
73 ) noexcept {
74 const double angle_d = angle_radians.to_double();
75 const double half_d = angle_d * 0.5;
76 const double s_d = std::sin(half_d);
77 const double c_d = std::cos(half_d);
78
79 const Scalar s = Scalar::from_double(s_d);
80 const Scalar c = Scalar::from_double(c_d);
81
82 return T81Quaternion(
83 c,
84 axis_x * s,
85 axis_y * s,
86 axis_z * s
87 );
88 }

127 // Hamilton product
128 [[nodiscard]] friend inline T81Quaternion operator*(
129 const T81Quaternion& a,
130 const T81Quaternion& b
131 ) noexcept {
132 // Let:
133 // a = (aw + ax i) + (ay j + az k)
134 // b = (bw + bx i) + (by j + bz k)
135 //
136 // Encoded as complex pairs:
137 // a.real_imag_ = aw + ax·i
138 // a.j_k_ = ay + az·i
139 // likewise for b.
140 const Complex ac = a.real_imag_ * b.real_imag_;
141 const Complex bd = a.j_k_ * b.j_k_.conj();
142 const Complex ab_cd = a.real_imag_ * b.j_k_;
143 const Complex cd_ab = a.j_k_ * b.real_imag_;
144
145 const Scalar w = ac.real() - bd.real();
146 const Scalar x = ac.imag() + bd.imag();
147 const Scalar y = ab_cd.real() + cd_ab.real();
148 const Scalar z = ab_cd.imag() - cd_ab.imag();
149
150 return T81Quaternion(w, x, y, z);
151 }

189 // Magnitude (Euclidean norm of the quaternion)
190 [[nodiscard]] inline Scalar norm() const noexcept {
191 const Scalar m2 = mag2();
192 const double d = m2.to_double();
193 if (d <= 0.0) {
194 return Scalar::from_double(0.0);
195 }
196 return Scalar::from_double(std::sqrt(d));
197 }

199 // Normalize (unit quaternion). If length is zero, returns identity().
200 [[nodiscard]] inline T81Quaternion normalized() const noexcept {
201 const Scalar m2 = mag2();
202 double d = m2.to_double();
203
204 if (d <= 0.0) {
205 return identity();
206 }
207
208 const double inv_len_d = 1.0 / std::sqrt(d);
209 const Scalar inv_len = Scalar::from_double(inv_len_d);
210 return *this * inv_len;
211 }

213 // Rotate a 3D vector.
214 //
215 // Interprets (vx, vy, vz) as a "pure" quaternion v = (0, vx, vy, vz),
216 // then returns q * v * q.conj(). The returned quaternion has w ≈ 0,
217 // and the rotated vector lives in {x, y, z}.
218 [[nodiscard]] inline T81Quaternion rotate_vector(
219 const Scalar& vx,
220 const Scalar& vy,
221 const Scalar& vz
222 ) const noexcept {
223 const T81Quaternion v(
224 Scalar::from_double(0.0),
225 vx,
226 vy,
227 vz
228 );
229 const T81Quaternion result = (*this) * v * this->conj();
230 return result;
231 }

268[[nodiscard]] inline T81Quaternion slerp(
269 const T81Quaternion& a,
270 const T81Quaternion& b,
271 const T81Quaternion::Scalar& t
272) noexcept {
273 using Scalar = T81Quaternion::Scalar;
274
275 // Convert components to double for transcendental math
276 const double aw = a.w().to_double();
277 const double ax = a.x().to_double();
278 const double ay = a.y().to_double();
279 const double az = a.z().to_double();
280
281 const double bw = b.w().to_double();
282 const double bx = b.x().to_double();
283 const double by = b.y().to_double();
284 const double bz = b.z().to_double();
285
286 // Cosine of the angle between them
287 double dot = aw * bw + ax * bx + ay * by + az * bz;
288
289 // If dot < 0, use the negated second quaternion to take the short arc
290 double bw_adj = bw;
291 double bx_adj = bx;
292 double by_adj = by;
293 double bz_adj = bz;
294
295 if (dot < 0.0) {
296 dot = -dot;
297 bw_adj = -bw;
298 bx_adj = -bx;
299 by_adj = -by;
300 bz_adj = -bz;
301 }
302
303 const double t_d = t.to_double();
304
305 // If very close, fall back to normalized lerp
306 if (dot > 0.9999) {
307 const double w = aw + (bw_adj - aw) * t_d;
308 const double x = ax + (bx_adj - ax) * t_d;
309 const double y = ay + (by_adj - ay) * t_d;
310 const double z = az + (bz_adj - az) * t_d;
311
312 T81Quaternion lerp(
313 Scalar::from_double(w),
314 Scalar::from_double(x),
315 Scalar::from_double(y),
316 Scalar::from_double(z)
317 );
318 return lerp.normalized();
319 }
320
321 // Clamp dot into [-1, 1] to protect acos
322 if (dot < -1.0) dot = -1.0;
323 if (dot > 1.0) dot = 1.0;
324
325 const double theta = std::acos(dot);
326 const double sin_theta = std::sin(theta);
327
328 if (sin_theta == 0.0) {
329 // Degenerate case; just return a
330 return a;
331 }
332
333 const double a_factor = std::sin((1.0 - t_d) * theta) / sin_theta;
334 const double b_factor = std::sin(t_d * theta) / sin_theta;
335
336 const double w = aw * a_factor + bw_adj * b_factor;
337 const double x = ax * a_factor + bx_adj * b_factor;
338 const double y = ay * a_factor + by_adj * b_factor;
339 const double z = az * a_factor + bz_adj * b_factor;
340
341 T81Quaternion result(
342 Scalar::from_double(w),
343 Scalar::from_double(x),
344 Scalar::from_double(y),
345 Scalar::from_double(z)
346 );
347 return result.normalized();
348}
