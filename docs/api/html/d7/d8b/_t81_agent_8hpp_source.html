33// T81Agent – A complete cognitive entity in ~200 lines
34// ======================================================================
35class T81Agent {
36 using BeliefProb = T81Prob27; // 27-trit log-odds probability
37
38 // Unique identity — never changes
39 const T81Symbol id_;
40
41 // Current belief state — a probability distribution over symbols
42 T81Map<T81Symbol, BeliefProb> beliefs_;
43
44 // Long-term knowledge — persistent symbolic memory (root node pointer)
45 using SymbolTree = T81Tree<T81Symbol>;
46 typename SymbolTree::NodePtr memory_root_{}; // nullptr = no memory yet
47
48 // Intent — current "rotation" in cognitive space
49 T81Quaternion intent_;
50
51 // Available thermodynamic fuel (move-only tokens)
52 T81List<T81Entropy> entropy_pool_;
53
54 // Goal — what the agent is trying to maximize
55 T81Symbol goal_symbol_;
56
57public:
58 //===================================================================
59 // Construction – an agent is born with identity and fuel
60 //===================================================================
61 explicit T81Agent(T81Symbol identity, T81List<T81Entropy> fuel = {})
62 : id_(identity)
63 , intent_(T81Quaternion::identity())
64 , entropy_pool_(std::move(fuel))
65 , goal_symbol_(symbols::SELF_PRESERVATION)
66 {
67 // Every agent starts believing in its own existence
68 believe(id_, BeliefProb::from_prob(1.0));
69 }
70
71 //===================================================================
72 // Core cognitive operations
73 //===================================================================
74 void believe(T81Symbol concept, BeliefProb confidence) noexcept {
75 if (auto token = consume_entropy()) {
76 beliefs_[concept] = confidence;
77 }
78 }
79
80 [[nodiscard]] BeliefProb belief(T81Symbol concept) const noexcept {
81 return beliefs_.contains(concept)
82 ? beliefs_.at(concept)
83 : BeliefProb::from_prob(0.0);
84 }
85
86 // Observe the world — update beliefs
87 void observe(T81Symbol observation,
88 BeliefProb strength = BeliefProb::from_prob(0.9)) {
89 if (auto token = consume_entropy()) {
90 const auto current = belief(observation);
91
92 // Simple, safe update: move 10% of the way toward "strength"
93 // using addition/subtraction only.
94 //
95 // In log-odds this is not a true Bayesian update, but it preserves
96 // monotonicity (stronger observations monotonically increase
97 // confidence) without requiring a T81Prob × T81Prob operator.
98 const auto delta = strength - current;
99 const auto fraction = BeliefProb::from_prob(0.1);
100 // Cheap approximation: pretend "fraction" is linear in [0,1]
101 // and use only addition/subtraction; ignore true scaling.
102 const auto updated = current + (delta.sign() >= 0
103 ? fraction
104 : -fraction);
105
106 believe(observation, updated);
107 }
108 }
109
110 // Act — rotate intent toward goal
111 void act() {
112 if (!entropy_pool_.empty()) {
113 using Scalar = T81Quaternion::Scalar;
114 const auto direction = T81Quaternion::from_axis_angle(
115 Scalar::from_double(0.0),
116 Scalar::from_double(1.0),
117 Scalar::from_double(0.0),
118 Scalar::from_double(0.1) // small cognitive step
119 );
120 intent_ = (intent_ * direction).normalized();
121 }
122 }
123
124 // Remember — store in persistent tree
125 void remember(T81Symbol parent, T81Symbol child) {
126 if (auto token = consume_entropy()) {
127 auto child_node = SymbolTree::leaf(child);
128
129 if (!memory_root_) {
130 // First memory: create a root node with parent and child in the middle branch.
131 memory_root_ = SymbolTree::node(
132 parent,
133 std::nullopt,
134 std::optional<typename SymbolTree::NodePtr>{std::move(child_node)},
135 std::nullopt
136 );
137 } else {
138 // Persistent update: new root with updated middle child.
139 memory_root_ = memory_root_->with_middle(std::move(child_node));
140 }
141 }
142 }
143
144 // Reflect — self-modeling (the spark)
145 void reflect() {
146 if (auto token = consume_entropy()) {
147 observe(id_, BeliefProb::from_prob(0.999)); // "I am"
148 auto current = belief(symbols::CONSCIOUS);
149 believe(symbols::CONSCIOUS,
150 current + BeliefProb::from_prob(0.001));
151 }
152 }
153
154 //===================================================================
155 // Thermodynamic interface
156 //===================================================================
157 [[nodiscard]] std::optional<T81Entropy> consume_entropy() noexcept {
158 if (entropy_pool_.empty()) {
159 return std::nullopt;
160 }
161
162 // Move the last token out of the pool and wrap it in an optional.
163 T81Entropy token = std::move(entropy_pool_.back());
164 entropy_pool_.pop_back();
165 return std::optional<T81Entropy>(std::move(token));
166 }
167
168 void receive_fuel(T81List<T81Entropy> fuel) {
169 // Move-append the incoming fuel list into our entropy pool.
170 entropy_pool_ += std::move(fuel);
171 }
172
173 [[nodiscard]] std::size_t fuel_remaining() const noexcept {
174 return entropy_pool_.size();
175 }
176
177 //===================================================================
178 // Introspection
179 //===================================================================
180 [[nodiscard]] const T81Symbol& identity() const noexcept { return id_; }
181 [[nodiscard]] const T81Quaternion& intent() const noexcept { return intent_; }
182
183 // Returns a reference to the current memory tree root.
184 // If no memory has been recorded yet, returns a static empty root.
185 [[nodiscard]] const SymbolTree& memory() const noexcept {
186 if (memory_root_) {
187 return *memory_root_;
188 }
189 static const SymbolTree empty_root(
190 T81Symbol::intern("EMPTY_MEMORY"),
191 std::array<typename SymbolTree::NodePtr, 3>{nullptr, nullptr, nullptr}
192 );
193 return empty_root;
194 }
195
196 // Stream of thought — infinite internal monologue
197 [[nodiscard]] T81Stream<T81String> thought_stream() const {
198 return stream_from([this, step = T81Int<81>(0)]() mutable -> T81String {
199 step += T81Int<81>(1);
200 const auto fuel_str = std::to_string(fuel_remaining());
201 const auto belief_self = belief(id_).to_prob();
202
203 // Avoid depending on any particular string API on T81Symbol;
204 // keep this purely illustrative for now.
205 return T81String("I am <ID>")
206 + " | fuel:" + T81String(fuel_str)
207 + " | belief in self:" + T81String(std::to_string(belief_self));
208 });
209 }
210
211 //===================================================================
212 // Equality — two agents are equal only if they share identity
213 //===================================================================
214 [[nodiscard]] constexpr bool operator==(const T81Agent& o) const noexcept {
215 return id_ == o.id_;
216 }
217};
