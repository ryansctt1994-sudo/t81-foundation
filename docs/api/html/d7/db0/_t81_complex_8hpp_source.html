120 [[nodiscard]] Float phase() const noexcept {
121 const double x = re.to_double();
122 const double y = im.to_double();
123
124 if (x == 0.0 && y == 0.0) {
125 return Float::from_double(0.0);
126 }
127
128 constexpr double kTwoPi = 6.28318530717958647692;
129 double angle = std::atan2(y, x);
130 double turns = angle / kTwoPi;
131 if (turns < 0.0) turns += 1.0;
132 return Float::from_double(turns);
133 }

186template <std::size_t M>
187[[nodiscard]] T81Complex<M> expi(typename T81Complex<M>::FloatType theta) noexcept {
188 constexpr double kTwoPi = 6.28318530717958647692;
189 const double angle = theta.to_double() * kTwoPi;
190 const double c = std::cos(angle);
191 const double s = std::sin(angle);
192 using Float = typename T81Complex<M>::FloatType;
193 return T81Complex<M>(Float::from_double(c), Float::from_double(s));
194}

196[[nodiscard]] inline T81Complex18 mul3(const T81Complex18& a, const T81Complex18& b) noexcept {
197 using Float = T81Complex18::Float;
198 const Float p = a.re * (b.re + b.im);
199 const Float q = b.re * (a.im + a.re);
200 const Float r = a.im * (b.im - b.re);
201 return T81Complex18{ p - q, p + r };
