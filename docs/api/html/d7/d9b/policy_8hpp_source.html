101class PolicyLexer {
102 public:
103 explicit PolicyLexer(std::string_view src) : src_(src) {}
104
105 PolicyToken next() {
106 skip_ws_();
107 if (pos_ >= src_.size()) return PolicyToken{};
108 char c = src_[pos_];
109 if (c == '(') { ++pos_; return PolicyToken{PolicyToken::Kind::LParen, {}}; }
110 if (c == ')') { ++pos_; return PolicyToken{PolicyToken::Kind::RParen, {}}; }
111 if (std::isdigit(static_cast<unsigned char>(c)) || c == '-' || c == '+') {
112 std::size_t start = pos_;
113 ++pos_;
114 while (pos_ < src_.size() &&
115 std::isdigit(static_cast<unsigned char>(src_[pos_]))) ++pos_;
116 PolicyToken tok;
117 tok.kind = PolicyToken::Kind::Integer;
118 tok.text = std::string(src_.substr(start, pos_ - start));
119 tok.value = std::stoll(tok.text);
120 return tok;
121 }
122 if (c == '"') {
123 ++pos_;
124 std::size_t start = pos_;
125 while (pos_ < src_.size() && src_[pos_] != '"') {
126 ++pos_;
127 }
128 PolicyToken tok;
129 tok.kind = PolicyToken::Kind::String;
130 tok.text = std::string(src_.substr(start, pos_ - start));
131 if (pos_ < src_.size() && src_[pos_] == '"') ++pos_;
132 return tok;
133 }
134 if (std::isalpha(static_cast<unsigned char>(c)) || c == '-' || c == '_') {
135 std::size_t start = pos_;
136 ++pos_;
137 while (pos_ < src_.size()) {
138 char ch = src_[pos_];
139 if (!(std::isalnum(static_cast<unsigned char>(ch)) || ch == '-' ||
140 ch == '_')) {
141 break;
142 }
143 ++pos_;
144 }
145 PolicyToken tok;
146 tok.kind = PolicyToken::Kind::Symbol;
147 tok.text = std::string(src_.substr(start, pos_ - start));
148 return tok;
149 }
150 // Unknown char -> skip and treat as End to force parse failure.
151 pos_ = src_.size();
152 return PolicyToken{};
153 }
154
155 private:
156 void skip_ws_() {
157 while (pos_ < src_.size() &&
158 std::isspace(static_cast<unsigned char>(src_[pos_]))) {
159 ++pos_;
160 }
161 }
162
163 std::string_view src_;
164 std::size_t pos_{0};
165};
