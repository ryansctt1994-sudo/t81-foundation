44// ======================================================================
45// T81Category – A category is a set of objects and morphisms
46// ======================================================================
47template <typename Obj, typename Mor>
48class T81Category {
49 T81Set<Obj> objects_;
50 T81Map<T81Symbol, T81Morphism<Obj,Mor>> morphisms_;
51 T81Map<Obj, T81Symbol> identity_map_; // identity morphism per object
52
53public:
54 using object_type = Obj;
55 using morphism_type = Mor;
56
57 //===================================================================
58 // Construction
59 //===================================================================
60 constexpr T81Category() = default;
61
62 // Add object
63 constexpr T81Category& add_object(Obj obj, T81Symbol identity_name = {}) {
64 objects_ = objects_.insert(obj);
65 if (identity_name) {
66 auto id_mor = T81Morphism<Obj,Mor>(identity_name, obj, obj, Mor{});
67 morphisms_[identity_name] = id_mor;
68 identity_map_[obj] = identity_name;
69 }
70 return *this;
71 }

73 // Add morphism
74 constexpr T81Category& add_morphism(T81Symbol name, Obj src, Obj dst, Mor data) {
75 if (!objects_.contains(src) || !objects_.contains(dst))
76 return *this; // invalid
77 morphisms_[name] = T81Morphism<Obj,Mor>(name, src, dst, std::move(data));
78 return *this;
79 }

81 //===================================================================
82 // Composition – the heart of category theory
83 //===================================================================
84 [[nodiscard]] std::optional<T81Morphism<Obj,Mor>> compose(
85 const T81Symbol& f_name,
86 const T81Symbol& g_name
87 ) const {
88 auto f_it = morphisms_.find(f_name);
89 auto g_it = morphisms_.find(g_name);
90 if (!f_it || !g_it) return std::nullopt;
91
92 const auto& f = *f_it;
93 const auto& g = *g_it;
94 if (!f.composable_with(g)) return std::nullopt;
95
96 // Compute composite data — depends on Mor type
97 Mor composite_data = compose_data(f.data, g.data);
98 T81Symbol composite_name = T81Symbol::intern(f.name.str() + " ∘ " + g.name.str());
99
100 return T81Morphism<Obj,Mor>(composite_name, g.source, f.target, std::move(composite_data));
101 }

103 //===================================================================
104 // Identity
105 //===================================================================
106 [[nodiscard]] constexpr T81Symbol identity_of(Obj obj) const {
107 return identity_map_.contains(obj) ? identity_map_.at(obj) : symbols::NO_IDENTITY;
108 }

116private:
117 // Default composition — override via specialization
118 static Mor compose_data(const Mor& f, const Mor& g) {
119 // Default: assume Mor supports composition (e.g., function, matrix)
120 if constexpr (requires { f(g); }) {
121 return f(g);
122 } else {
123 static_assert(std::is_same_v<Mor, void>, "Specialize compose_data for this morphism type");
124 return Mor{};
125 }

138
139public:

142
143 constexpr T81Functor& map_object(typename C::object_type src, typename D::object_type dst) {
144 object_map_[src] = dst;
145 return *this;
