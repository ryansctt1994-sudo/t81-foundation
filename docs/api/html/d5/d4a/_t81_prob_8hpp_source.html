30// ======================================================================
31// T81Prob<Trits> — Native log-odds / log-probability
32// ======================================================================
33//
34// Representation:
35// Stored as log-odds in base φ ≈ 1.618 (golden ratio) or natural log.
36// Trits controls the underlying T81Int width:
37// Trits = 27 → ~42.8 bits of precision → vastly superior to FP16 log-probs
38// Exact representation of many structured probabilities (powers of 1/3, 1/9,…).
39// Softmax → ternary addition (no exp, no div, no overflow in the hot path).
40// Sampling / arithmetic coding can be layered on top later.
41//
42// NOTE: The scale (kScale) is currently independent of Trits. Wider Trits simply
43// allow a larger dynamic range of log-odds before saturation.
44// ======================================================================
45template <std::size_t Trits>
46class T81Prob {
47public:
48 using Storage = T81Int<Trits>;
49
50private:
51 Storage log_odds_{}; // log(p / (1-p)) in fixed-point base-φ units
52
53public:
54 static constexpr std::size_t kTrits = Trits;
55
56 // ------------------------------------------------------------------
57 // Constructors
58 // ------------------------------------------------------------------
59
60 T81Prob() noexcept = default;
61
62 explicit T81Prob(Storage v) noexcept : log_odds_(v) {}
63
64 // ------------------------------------------------------------------
65 // Core accessors
66 // ------------------------------------------------------------------
67
68 [[nodiscard]] const Storage& raw() const noexcept {
69 return log_odds_;
70 }
71
72 [[nodiscard]] const Storage& log_odds() const noexcept {
73 return log_odds_;
74 }

76 // ------------------------------------------------------------------
77 // Construction from real probability [0,1]
78 // ------------------------------------------------------------------
79 //
80 // Not constexpr: depends on libm (log / exp / llround).
81 //
82 [[nodiscard]] static T81Prob from_prob(double p) noexcept {
83 if (p <= 0.0) return minus_infinity();
84 if (p >= 1.0) return plus_infinity();
85 if (p == 0.5) return zero();
86
87 const double odds = p / (1.0 - p);
88 const double log_odds = std::log(odds);
89
90 // Scale to fixed-point in base φ ≈ 1.618
91 constexpr double phi = 1.6180339887498948482;
92 constexpr double kScale = 512.0; // ~9 "fractional trits" in this scheme
93
94 const double scaled = log_odds / std::log(phi);
95 const std::int64_t fixed =
96 static_cast<std::int64_t>(std::llround(scaled * kScale));
97
98 return T81Prob(Storage(fixed));
99 }
100
101 // ------------------------------------------------------------------
102 // Special values
103 // ------------------------------------------------------------------
104
105 // log-odds = 0 → p = 0.5
106 [[nodiscard]] static T81Prob zero() noexcept {
107 return T81Prob(Storage(0));
108 }
109
110 // One "unit" of log-odds (≈ 1.0) → p ≈ 0.73111
111 [[nodiscard]] static T81Prob one() noexcept {
112 return from_prob(0.731111);
113 }
114
115 [[nodiscard]] static T81Prob minus_infinity() noexcept {
116 return T81Prob(Storage::kMinValue);
117 }
118
119 [[nodiscard]] static T81Prob plus_infinity() noexcept {
120 return T81Prob(Storage::kMaxValue);
121 }
122
123 // ------------------------------------------------------------------
124 // Conversion back to probability
125 // ------------------------------------------------------------------
126
127 [[nodiscard]] double to_prob() const noexcept {
128 if (is_minus_infinity()) return 0.0;
129 if (is_plus_infinity()) return 1.0;
130
131 constexpr double phi = 1.6180339887498948482;
132 constexpr double kScale = 512.0;
133
134 const double scaled =
135 static_cast<double>(log_odds_.to_int64()) / kScale;
136 const double log_odds_real = scaled * std::log(phi);
137 const double odds = std::exp(log_odds_real);
138
139 return odds / (1.0 + odds);
140 }
141
142 // ------------------------------------------------------------------
143 // Queries
144 // ------------------------------------------------------------------
145
146 [[nodiscard]] bool is_zero() const noexcept {
147 return log_odds_.is_zero();
148 }
149
150 [[nodiscard]] bool is_minus_infinity() const noexcept {
151 return log_odds_ == Storage::kMinValue;
152 }
153
154 [[nodiscard]] bool is_plus_infinity() const noexcept {
155 return log_odds_ == Storage::kMaxValue;
156 }
157
158 // ------------------------------------------------------------------
159 // Core arithmetic — THIS IS WHY IT'S MAGIC
160 // ------------------------------------------------------------------
161
162 [[nodiscard]] T81Prob operator+(const T81Prob& o) const noexcept {
163 return T81Prob(log_odds_ + o.log_odds_);
164 }
165
166 [[nodiscard]] T81Prob operator-(const T81Prob& o) const noexcept {
167 return T81Prob(log_odds_ - o.log_odds_);
168 }
169
170 [[nodiscard]] T81Prob operator-() const noexcept {
171 return T81Prob(Storage(0) - log_odds_);
172 }
173
174 // Softmax over a tensor becomes: just add all → subtract each
175 // log_softmax(x_i) = x_i - log_sum_exp(x)
176 // → in T81Prob: x_i + (-log_sum_exp_all)
177 [[nodiscard]] T81Prob
178 log_softmax_normalize(const T81Prob& log_sum_exp) const noexcept {
179 return *this - log_sum_exp;
180 }
181

183 // Comparison
184 // ------------------------------------------------------------------

186 [[nodiscard]] auto operator<=>(const T81Prob& o) const noexcept = default;
187 [[nodiscard]] bool operator==(const T81Prob& o) const noexcept = default;
188};

204// Log-sum-exp over a span (fused into one ternary reduction on Axion)
205[[nodiscard]] inline T81Prob27
206log_sum_exp(std::span<const T81Prob27> probs) noexcept {
207 if (probs.empty()) {
208 return T81Prob27::minus_infinity();
209 }
210
211 const auto max_it = std::max_element(probs.begin(), probs.end());
212 const T81Prob27 max = *max_it;
213
214 T81Prob27 sum = T81Prob27::zero();
215 for (const auto& p : probs) {
216 sum = sum + (p - max);
217 }
218 return max + sum;
219}
