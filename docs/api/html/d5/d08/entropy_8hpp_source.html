13// --- Simple Shannon entropy (bits per symbol) over bytes ---
14inline double shannon_bits_per_byte(const std::vector<uint8_t>& data) {
15 if (data.empty()) return 0.0;
16 std::array<uint64_t, 256> freq{};
17 for (uint8_t b : data) ++freq[b];
18
19 const double n = static_cast<double>(data.size());
20 double H = 0.0;
21 for (uint64_t f : freq) {
22 if (!f) continue;
23 const double p = static_cast<double>(f) / n;
24 H -= p * std::log2(p);
25 }
26 return H; // in [0,8]
27}
