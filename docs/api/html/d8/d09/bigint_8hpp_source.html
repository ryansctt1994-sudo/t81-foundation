53 static T81BigInt from_base81_digit_string(std::string_view s) {
54 if (s.empty()) throw std::invalid_argument("T81BigInt::from_base81_digit_string: empty input");
55
56 bool neg = false;
57 std::size_t pos = 0;
58 if (s[pos] == '+' || s[pos] == '-') {
59 neg = (s[pos] == '-');
60 ++pos;
61 if (pos >= s.size()) throw std::invalid_argument("T81BigInt::from_base81_digit_string: sign only");
62 }
63
64 std::vector<int> digits;
65 int current = 0;
66 bool have_digit = false;
67
68 for (; pos <= s.size(); ++pos) {
69 if (pos == s.size() || s[pos] == '.') {
70 if (!have_digit) throw std::invalid_argument("T81BigInt::from_base81_digit_string: empty digit");
71 if (current < 0 || current >= kRadix) throw std::invalid_argument("T81BigInt::from_base81_digit_string: digit out of range");
72 digits.push_back(current);
73 current = 0;
74 have_digit = false;
75 } else if (s[pos] >= '0' && s[pos] <= '9') {
76 have_digit = true;
77 current = current * 10 + (s[pos] - '0');
78 if (current >= kRadix) throw std::invalid_argument("T81BigInt::from_base81_digit_string: digit overflow");
79 } else {
80 throw std::invalid_argument("T81BigInt::from_base81_digit_string: invalid character");
81 }
82 }
83
84 // Normalize leading zeros (except lone zero)
85 std::size_t first_nonzero = 0;
86 while (first_nonzero < digits.size() - 1 && digits[first_nonzero] == 0) ++first_nonzero;
87
88 T81BigInt out;
89 out.sign_ = Sign::Pos;
90 out.d_.clear();
91
92 const int base = kRadix;
93 for (std::size_t i = first_nonzero; i < digits.size(); ++i) {
94 mul_small_(out.d_, base, out.d_);
95 add_small_(out.d_, static_cast<uint16_t>(digits[i]), out.d_);
96 }
97
98 out.trim_();
99 if (out.is_zero()) {
100 return zero();
101 }
102 if (neg) out.sign_ = Sign::Neg;
103 return out;
104 }

148 if (sign_ == Sign::Zero) return 0;

152 const uint64_t negative_limit = static_cast<uint64_t>(INT64_MAX) + 1;
153 const uint64_t limit = (sign_ == Sign::Pos) ? positive_limit : negative_limit;

176 static T81BigInt add(const T81BigInt& a, const T81BigInt& b) {
177 if (is_zero(a)) return b;
178 if (is_zero(b)) return a;

203 // Very naive modulus: repeated subtraction by shifted divisor (long division skeleton).
204 // Works but is slow; only intended for tiny values / tests.
205 static T81BigInt mod(const T81BigInt& a, const T81BigInt& b) {
206 if (is_zero(b)) throw std::domain_error("BigInt mod by zero");
207 if (is_zero(a)) return zero();

218 static T81BigInt gcd(T81BigInt a, T81BigInt b) {

224 a = std::move(b);

230 static T81BigInt pow(const T81BigInt& base, const T81BigInt& exp) {
231 if (exp.is_negative()) {
232 throw std::domain_error("BigInt pow: negative exponent");
233 }
234 if (is_zero(exp)) return one();
235 if (is_zero(base)) return zero();
236
237 T81BigInt result = one();
238 T81BigInt b = abs(base);
239 T81BigInt e = exp;
240 const T81BigInt two(2);
241
242 while (e > zero()) {
243 if (!is_zero(mod(e, two))) { // if e is odd
244 result = mul(result, b);
245 }
246 b = mul(b, b);
247 T81BigInt q, r;
248 divmod_nonneg_(e, two, q, r);
249 e = q;
250 }
251
252 if (base.is_negative() && !is_zero(mod(exp, two))) { // if base is negative and exp is odd
253 return neg(result);
254 }
255 return result;
256 }

330 static void add_mag_(const std::vector<uint8_t>& A, const std::vector<uint8_t>& B, std::vector<uint8_t>& out) {
331 const size_t n = std::max(A.size(), B.size());
332 out.clear(); out.resize(n, 0);
333 uint16_t carry = 0;
334 for (size_t i = 0; i < n; ++i) {
335 uint16_t a = (i < A.size() ? A[i] : 0);
336 uint16_t b = (i < B.size() ? B[i] : 0);
337 uint16_t s = a + b + carry;
338 out[i] = static_cast<uint8_t>(s % kRadix);
339 carry = static_cast<uint16_t>(s / kRadix);
340 }
341 if (carry) out.push_back(static_cast<uint8_t>(carry));
342 }

344 static void sub_mag_(const std::vector<uint8_t>& A, const std::vector<uint8_t>& B, std::vector<uint8_t>& out) {
345 // assumes A >= B
346 out.clear(); out.resize(A.size(), 0);
347 int16_t borrow = 0;
348 for (size_t i = 0; i < A.size(); ++i) {
349 int16_t a = A[i];
350 int16_t b = (i < B.size() ? B[i] : 0);
351 int16_t s = a - b - borrow;
352 if (s < 0) { s += kRadix; borrow = 1; } else borrow = 0;
353 out[i] = static_cast<uint8_t>(s);
354 }
355 // trim handled by caller
356 while (!out.empty() && out.back()==0) out.pop_back();
357 }

359 static void schoolbook_mul_(const std::vector<uint8_t>& A, const std::vector<uint8_t>& B, std::vector<uint8_t>& out) {
360 out.assign(A.size()+B.size(), 0);
361 for (size_t i = 0; i < A.size(); ++i) {
362 uint32_t carry = 0;
363 for (size_t j = 0; j < B.size(); ++j) {
364 uint32_t s = out[i+j] + uint32_t(A[i]) * uint32_t(B[j]) + carry;
365 out[i+j] = static_cast<uint8_t>(s % kRadix);
366 carry = static_cast<uint32_t>(s / kRadix);
367 }
368 size_t k = i + B.size();
369 while (carry) {
370 uint32_t s = out[k] + carry;
371 out[k] = static_cast<uint8_t>(s % kRadix);
372 carry = static_cast<uint32_t>(s / kRadix);
373 ++k;
374 }
375 }
376 while (!out.empty() && out.back()==0) out.pop_back();
377 }

379 static void mul_small_(const std::vector<uint8_t>& A, uint16_t m, std::vector<uint8_t>& out) {
380 if (m == 0 || A.empty()) { out.clear(); return; }
381 if (m == 1) { out = A; return; }
382 out.resize(A.size());
383 uint32_t carry = 0;
384 for (size_t i = 0; i < A.size(); ++i) {
385 uint32_t s = uint32_t(A[i]) * m + carry;
386 out[i] = static_cast<uint8_t>(s % kRadix);
387 carry = static_cast<uint32_t>(s / kRadix);
388 }
389 while (carry) {
390 out.push_back(static_cast<uint8_t>(carry % kRadix));
391 carry /= kRadix;
392 }
393 while (!out.empty() && out.back()==0) out.pop_back();
394 }

396 static void add_small_(const std::vector<uint8_t>& A, uint16_t m, std::vector<uint8_t>& out) {
397 out = A;
398 uint32_t s = (out.empty() ? 0u : out[0]) + m;
399 if (out.empty()) out.push_back(0);
400 out[0] = static_cast<uint8_t>(s % kRadix);
401 uint32_t carry = s / kRadix;
402 size_t i = 1;
403 while (carry) {
404 if (i >= out.size()) out.push_back(0);
405 uint32_t t = out[i] + carry;
406 out[i] = static_cast<uint8_t>(t % kRadix);
407 carry = t / kRadix;
408 ++i;
409 }
410 while (!out.empty() && out.back()==0) out.pop_back();
411 }

413 // Slow but simple non-negative divmod used by stubs/tests.
414 static void divmod_nonneg_(const T81BigInt& ua, const T81BigInt& ub,
415 T81BigInt& uq, T81BigInt& ur) {
416 if (is_zero(ub)) throw std::domain_error("BigInt div by zero");
417 uq = zero();
418 ur = ua;
419 if (cmp(ua, ub) < 0) return;
420 const T81BigInt one_val = one();
421
422 while (cmp(ur, ub) >= 0) {
423 T81BigInt multiple = ub;
424 T81BigInt factor = one_val;
425 while (cmp(add(multiple, multiple), ur) <= 0) {
426 multiple = add(multiple, multiple);
427 factor = add(factor, factor);
428 }
429 ur = sub(ur, multiple);
430 uq = add(uq, factor);
431 }
432 }

465inline std::string next_codepoint(std::string_view s, std::size_t& offset) {
466 if (offset >= s.size()) return {};
467 const unsigned char c = static_cast<unsigned char>(s[offset]);
468 std::size_t len = 0;
469 if (c < 0x80) len = 1;
470 else if ((c & 0xE0) == 0xC0) len = 2;
471 else if ((c & 0xF0) == 0xE0) len = 3;
472 else if ((c & 0xF8) == 0xF0) len = 4;
473 else return {};
474 if (offset + len > s.size()) return {};
475 std::string cp(s.substr(offset, len));
476 offset += len;
477 return cp;
478}

506inline T81BigInt T81BigInt::from_base81_string(std::string_view s) {
507 if (s.empty()) throw std::invalid_argument("T81BigInt::from_base81_string: empty input");
508
509 bool neg = false;
510 std::size_t pos = 0;
511 if (s[pos] == '+' || s[pos] == '-') {
512 neg = (s[pos] == '-');
513 ++pos;
514 if (pos >= s.size()) throw std::invalid_argument("T81BigInt::from_base81_string: sign only");
515 }
516
517 std::vector<int> digits;
518 const auto& map = detail::base81_digit_map();
519 while (pos < s.size()) {
520 std::string cp = detail::next_codepoint(s, pos);
521 if (cp.empty()) throw std::invalid_argument("T81BigInt::from_base81_string: invalid encoding");
522 auto it = map.find(cp);
523 if (it == map.end()) throw std::invalid_argument("T81BigInt::from_base81_string: invalid character");
524 digits.push_back(it->second);
525 }
526 if (digits.size() > 1 && digits.front() == 0) {
527 throw std::invalid_argument("T81BigInt::from_base81_string: non-canonical leading zero");
528 }
529
530 T81BigInt base(kRadix);
531 T81BigInt v(0);
532 for (int d : digits) {
533 v = mul(v, base);
534 v = add(v, T81BigInt(d));
535 }
536 if (neg && !is_zero(v)) v.sign_ = Sign::Neg;
537 return v;
538}
