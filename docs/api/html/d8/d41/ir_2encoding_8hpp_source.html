11// Fixed 32-byte little-endian layout:
12//
13// 0x00: uint16 op
14// 0x02: uint16 pad (0)
15// 0x04: uint32 ops[0]
16// 0x08: uint32 ops[1]
17// 0x0C: uint32 ops[2]
18// 0x10: uint64 imm
19// 0x18: uint32 flags
20// 0x1C: uint32 _reserved (0)
21//
22// All fields are encoded little-endian regardless of host endianness.

24inline void encode(const Insn& i, uint8_t out[32]) {
25 std::memset(out, 0, 32);
26 auto wr16 = [&](size_t off, uint16_t v){
27 out[off+0] = static_cast<uint8_t>( v & 0xFF);
28 out[off+1] = static_cast<uint8_t>((v >> 8) & 0xFF);
29 };
30 auto wr32 = [&](size_t off, uint32_t v){
31 out[off+0] = static_cast<uint8_t>( v & 0xFF);
32 out[off+1] = static_cast<uint8_t>((v >> 8) & 0xFF);
33 out[off+2] = static_cast<uint8_t>((v >> 16) & 0xFF);
34 out[off+3] = static_cast<uint8_t>((v >> 24) & 0xFF);
35 };
36 auto wr64 = [&](size_t off, uint64_t v){
37 for (int k = 0; k < 8; ++k) out[off+k] = static_cast<uint8_t>((v >> (8*k)) & 0xFF);
38 };

41 wr32(0x04, i.ops[0]);
42 wr32(0x08, i.ops[1]);

44 wr64(0x10, i.imm);

47}

49inline Insn decode(const uint8_t in[32]) {
50 auto rd16 = [&](size_t off)->uint16_t{
51 return static_cast<uint16_t>(in[off+0]) |
52 static_cast<uint16_t>(in[off+1]) << 8;
53 };
54 auto rd32 = [&](size_t off)->uint32_t{
55 return (uint32_t)in[off+0]
56 | ((uint32_t)in[off+1] << 8)
57 | ((uint32_t)in[off+2] << 16)
58 | ((uint32_t)in[off+3] << 24);
59 };
60 auto rd64 = [&](size_t off)->uint64_t{
61 uint64_t v = 0;
62 for (int k = 0; k < 8; ++k) v |= (uint64_t)in[off+k] << (8*k);
63 return v;
64 };
65
66 Insn i;
67 i.op = static_cast<Opcode>(rd16(0x00));
68 i.ops[0] = rd32(0x04);
69 i.ops[1] = rd32(0x08);
70 i.ops[2] = rd32(0x0C);
71 i.imm = rd64(0x10);
72 i.flags = rd32(0x18);
73 i._reserved = 0;
74 return i;
75}
