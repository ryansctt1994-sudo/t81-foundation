32// ======================================================================
33// Concept: any type that fits in one tryte (81 trits) or is void (for views)
34// ======================================================================
35template <typename T>
36concept T81Element =
37 sizeof(T) <= 32 && // ≤ 81 trits
38 (std::same_as<T, T81Int<81>> ||
39 std::same_as<T, T81Float<72,9>> || // 81-trit float
40 std::same_as<T, T81Fixed<72,9>> ||
41 std::same_as<T, T81Complex<40>> || // 80 trits → padded
42 std::same_as<T, T81Symbol>);

105 //===================================================================
111 template <size_t... NewDims>
112 requires (sizeof...(NewDims) == Rank)
113 [[nodiscard]] constexpr auto broadcast_to() const noexcept
114 -> T81Tensor<Element, Rank, NewDims...>
115 {
116 T81Tensor<Element, Rank, NewDims...> out;
117 std::array<size_t, Rank> old_shape = shape();
118 std::array<size_t, Rank> new_shape = {NewDims...};
119
120 // Simple tiling broadcast logic
121 for (size_t i = 0; i < out.size(); ++i) {
122 std::array<size_t, Rank> coords;
123 size_t temp_i = i;
124 for (int r = Rank - 1; r >= 0; --r) {
125 coords[r] = (temp_i % new_shape[r]) % old_shape[r];
126 temp_i /= new_shape[r];
127 }
128
129 size_t old_flat = 0;
130 size_t stride = 1;
131 for (int r = Rank - 1; r >= 0; --r) {
132 old_flat += coords[r] * stride;
133 stride *= old_shape[r];
134 }
135 out.span()[i] = span()[old_flat];
136 }
137 return out;
138 }

174 // Row-major linear index
175 template <typename... Indices>
176 [[nodiscard]] constexpr size_t linear_index(Indices... indices) const noexcept {
177 size_t idx[Rank] = { static_cast<size_t>(indices)... };
178 size_t flat = 0;
179 size_t stride = 1;
180 for (int i = static_cast<int>(Rank) - 1; i >= 0; --i) {
181 flat += idx[i] * stride;
182 stride *= shape()[static_cast<size_t>(i)];
183 }
184 return flat;
185 }

296template <typename E, size_t... Dims>
297[[nodiscard]] constexpr auto transpose(
298 const T81Tensor<E, sizeof...(Dims), Dims...>& t
299) noexcept {
300 constexpr size_t Rank = sizeof...(Dims);
301 constexpr std::array<size_t, Rank> old_shape = {Dims...};
302
303 // Create new shape by reversing old shape
304 auto reverse_shape = []() {
305 std::array<size_t, Rank> out{};
306 for (size_t i = 0; i < Rank; ++i) out[i] = old_shape[Rank - 1 - i];
307 return out;
308 }();
309
310 // We can't easily return a reversed type here because T81Tensor uses variadic Dims.
311 // In C++, return type must be fixed.
312 // If we want arbitrary rank transpose, we'd need a helper to generate the type.
313 // For now, we'll specialize common cases or use a placeholder that returns the same type
314 // ONLY if it's symmetric, but that's not useful.
315
316 return t; // placeholder for arbitrary rank
317}
