29template <std::size_t N, typename Scalar = T81Float<72,9>>
30class T81Vector {

35 // Helper: construct Scalar from double when available, otherwise via cast
36 static Scalar scalar_from_double(double v) noexcept {
37 if constexpr (requires { Scalar::from_double(v); }) {
38 return Scalar::from_double(v);
39 } else {
40 return static_cast<Scalar>(v);
41 }
42 }
43
44 // Bridge arbitrary component/scalar types to Scalar
45 template <typename C>
46 static constexpr Scalar component_to_scalar(const C& c) noexcept {
47 using Decayed = std::decay_t<C>;
48 if constexpr (std::same_as<Decayed, Scalar>) {
49 // Exact same type: copy
50 return c;
51 } else if constexpr (std::convertible_to<C, Scalar>) {
52 // Implicitly convertible (e.g., T81Float<18,9> → T81Float<72,9> via widening ctor)
53 return static_cast<Scalar>(c);
54 } else if constexpr (requires (const C& x) { { x.to_double() } -> std::convertible_to<double>; }) {
55 // Has to_double(): bridge via double → Scalar
56 return scalar_from_double(c.to_double());
57 } else {
58 static_assert(std::same_as<C, void>,
59 "T81Vector component type is not convertible or bridgeable to Scalar");
60 }
61 }

167 [[nodiscard]] constexpr Scalar length2() const noexcept { return dot(*this); }

169 [[nodiscard]] Scalar length() const noexcept {
170 if constexpr (requires (const Scalar& s) { { s.to_double() } -> std::convertible_to<double>; }) {
171 const double l2 = length2().to_double();
172 if (l2 <= 0.0) {
173 return scalar_from_double(0.0);
174 }
175 return scalar_from_double(std::sqrt(l2));
176 } else {
177 // Fallback: no bridge available, return squared length
178 return length2();
179 }
180 }
181
182 [[nodiscard]] T81Vector normalized() const noexcept {
183 if constexpr (requires (const Scalar& s) { { s.to_double() } -> std::convertible_to<double>; }) {
184 const double l2 = length2().to_double();
185 if (l2 <= 0.0) {
186 return *this;
187 }
188 const double len = std::sqrt(l2);
189 const double inv_d = 1.0 / len;
190 const Scalar inv_sc = scalar_from_double(inv_d);
191 return *this * inv_sc;
192 } else {
193 return *this;
194 }
195 }

197 // Cross product – only for 3D
198 template <std::size_t M = N>
199 [[nodiscard]] std::enable_if_t<M == 3, T81Vector> cross(const T81Vector& o) const noexcept {
200 return T81Vector(
201 components_[1] * o[2] - components_[2] * o[1],
202 components_[2] * o[0] - components_[0] * o[2],
203 components_[0] * o[1] - components_[1] * o[0]
204 );
205 }

207 // Angle between vectors
208 [[nodiscard]] Scalar angle(const T81Vector& o) const noexcept {
209 if constexpr (requires (const Scalar& s) { { s.to_double() } -> std::convertible_to<double>; }) {
210 const double l1 = length().to_double();
211 const double l2 = o.length().to_double();
212 double denom = l1 * l2;
213 if (denom <= 0.0) {
214 return scalar_from_double(0.0);
215 }
216 double c = dot(o).to_double() / denom;
217 if (c < -1.0) c = -1.0;
218 if (c > 1.0) c = 1.0;
219 return scalar_from_double(std::acos(c));
220 } else {
221 return scalar_from_double(0.0);
222 }
223 }

225 //===================================================================
226 // Rotation (using T81Quaternion)
227 //===================================================================
228 template <std::size_t M = N>
229 [[nodiscard]] std::enable_if_t<M == 3, T81Vector> rotated(const T81Quaternion& q) const noexcept {
230 using QScalar = T81Quaternion::Scalar;
231
232 // Bridge vector components into quaternion scalar space
233 const QScalar qx = QScalar::from_double(components_[0].to_double());
234 const QScalar qy = QScalar::from_double(components_[1].to_double());
235 const QScalar qz = QScalar::from_double(components_[2].to_double());
236
237 T81Quaternion vq(
238 QScalar::from_double(0.0),
239 qx, qy, qz
240 );
241 T81Quaternion result = q * vq * q.conj();
242
243 // Bridge back into this vector's scalar space
244 const Scalar rx = scalar_from_double(result.x().to_double());
245 const Scalar ry = scalar_from_double(result.y().to_double());
246 const Scalar rz = scalar_from_double(result.z().to_double());
247
248 return T81Vector(rx, ry, rz);
249 }
