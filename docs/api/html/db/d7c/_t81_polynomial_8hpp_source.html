89 // Evaluation – Horner's method (exact)
90 //===================================================================
91 [[nodiscard]] constexpr Coeff eval(const Coeff& x) const noexcept {
92 Coeff result(0);
93 for (auto it = coeffs_.rbegin(); it != coeffs_.rend(); ++it)
94 result = result * x + *it;

123 [[nodiscard]] friend constexpr T81Polynomial operator*(
124 const T81Polynomial& a,
125 const T81Polynomial& b
126 ) noexcept {
127 if (a.degree() == 0) return b * a[0];
128 if (b.degree() == 0) return a * b[0];
129
130 T81List<Coeff> c(a.degree() + b.degree() + 1, Coeff(0));
131 for (size_t i = 0; i <= a.degree(); ++i)
132 for (size_t j = 0; j <= b.degree(); ++j)
133 c[i + j] += a[i] * b[j];
134 return T81Polynomial(std::move(c));
135 }

151 [[nodiscard]] constexpr Division div(const T81Polynomial& divisor) const {
152 if (divisor.degree() == 0) {
153 return {*this * (Coeff(1) / divisor[0]), T81Polynomial{}};
154 }
155 if (degree() < divisor.degree()) {
156 return {{}, *this};
157 }
158
159 auto q = *this;
160 T81Polynomial r;
161 while (q.degree() >= divisor.degree()) {
162 auto lead = q.coefficients().back() / divisor.coefficients().back();
163 size_t deg_diff = q.degree() - divisor.degree();
164 auto term = T81Polynomial::monomial(deg_diff) * lead;
165 q = q - (divisor * term);
166 r = r + term;
167 }
168 r.trim();
169 return {r, q};
170 }
