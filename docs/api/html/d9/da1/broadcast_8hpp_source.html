9// NumPy-style right-aligned broadcasting.
10// Repeats elements of `src` to match `new_shape`.
11// Example: {3} -> {2,3}, {1,3} -> {4,3}, {2,1,4} -> {2,3,4}.
12template <typename T>
13inline T729TensorBase<T> broadcast_to(const T729TensorBase<T>& src, const std::vector<int>& new_shape) {
14 if (new_shape.empty()) throw std::invalid_argument("broadcast_to: empty new_shape");
15 // Build the right-aligned view of src.shape() against new_shape.
16 const auto& a = src.shape();
17 if (!t81::shape::can_broadcast_to(a, new_shape)) {
18 throw std::invalid_argument("broadcast_to: incompatible shapes");
19 }
20
21 // Materialize an aligned "src shape" padded with leading 1s to rank(new_shape)
22 const int r_out = static_cast<int>(new_shape.size());
23 std::vector<int> src_aligned(r_out, 1);
24 for (int i = 0; i < static_cast<int>(a.size()); ++i) {
25 src_aligned[static_cast<size_t>(r_out - 1 - i)] = a[static_cast<size_t>(a.size() - 1 - i)];
26 }
27
28 // Precompute strides (row-major) for both shapes.
29 const auto in_strides = t81::shape::strides_of(src_aligned);
30 const auto out_strides = t81::shape::strides_of(new_shape);
31
32 // Total output elements.
33 const std::size_t out_sz = t81::shape::size_of(new_shape);
34
35 std::vector<T> out(out_sz);
36 const auto& din = src.data();
37
38 // Iterate flat over output, remap to input index (clamp broadcasted dims to 0).
39 std::vector<int> idx(static_cast<size_t>(r_out), 0);
40 for (std::size_t flat = 0; flat < out_sz; ++flat) {
41 // Decode flat index into multi-index for out shape.
42 std::size_t rem = flat;
43 for (int d = 0; d < r_out; ++d) {
44 idx[static_cast<size_t>(d)] = static_cast<int>(rem / out_strides[static_cast<size_t>(d)]);
45 rem %= out_strides[static_cast<size_t>(d)];
46 }
47 // Map to input flat index.
48 std::size_t in_flat = 0;
49 for (int d = 0; d < r_out; ++d) {
50 const int dim_src = src_aligned[static_cast<size_t>(d)];
51 const int ii = (dim_src == 1) ? 0 : idx[static_cast<size_t>(d)];
52 in_flat += static_cast<std::size_t>(ii) * in_strides[static_cast<size_t>(d)];
53 }
54 out[flat] = din[in_flat];
55 }
56
57 return T729TensorBase<T>(new_shape, std::move(out));
58}
