51 //===================================================================
52 // Spawn a new thread of consciousness
53 //===================================================================
54 template <typename F>
55 [[nodiscard]] static T81Thread spawn(T81Symbol name, T81Agent thinker, T81Entropy fuel, F&& task) {
56 auto state = std::make_shared<ThreadState>(std::move(thinker), fuel, name);
57
58 state->handle = std::thread([state, task = std::forward<F>(task)]() mutable {
59 state->self.observe(symbols::I_AM_ALIVE);
60 state->self.observe(state->name);
61
62 while (void)task; // execute user's task
63
64 // Task may return a promise, or just run
65 try {
66 if constexpr (requires { task(); }) {
67 auto result = task();
68 if constexpr (requires { result.await(T81Entropy{}, state->self); }) {
69 while (!result.try_get()) {
70 if (!state->alive.load()) break;
71 std::this_thread::sleep_for(std::chrono::milliseconds(1));
72 consume_entropy(T81Entropy::acquire());
73 }
74 }
75 }
76 } catch (...) {
77 state->self.observe(symbols::THREAD_PANIC);
78 }
79
80 state->alive = false;
81 record_event(T81Time::now(T81Entropy::acquire(), symbols::THREAD_DEATH));
82 });
83
84 record_event(state->born_at);
85 return T81Thread(state);
86 }
