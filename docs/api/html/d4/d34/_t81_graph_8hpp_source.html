109 [[nodiscard]] friend constexpr auto pagerank(const T81Graph& g, int steps = 20) noexcept
110 -> T81Tensor<Weight81, 1, NodeCount>
111 {
112 T81Tensor<Weight81, 2, NodeCount, NodeCount> A{};
113 for (NodeID i = 0; i < NodeCount; ++i) {
114 size_t deg = g.outgoing(i).size();
115 Weight81 p = deg ? Weight81(1) / Weight81(deg) : Weight81(0);
116 for (auto [j, w] : g.outgoing(i)) {
117 A(i,j) = w * p;
118 }
119 }
120 // Add teleportation
121 for (size_t i = 0; i < NodeCount; ++i)
122 for (size_t j = 0; j < NodeCount; ++j)
123 A(i,j) = A(i,j) * Weight81(0.85) + Weight81(0.15 / NodeCount);
124
125 auto v = T81Tensor<Weight81, 1, NodeCount>::zeros();
126 v(0,0) = Weight81(1); // initial state
127
128 for (int i = 0; i < steps; ++i) v = v * A;
129 return v;
130 }

134 const T81Tensor<Weight81, 1, NodeCount>& node_states) const noexcept
