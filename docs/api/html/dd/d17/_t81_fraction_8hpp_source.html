51 // Reduce fraction and ensure positive denominator.
52 void canonicalize() noexcept {
53 // Normalize zero denominator to 0/1 deterministically.
54 if (den_.is_zero()) {
55 num_ = Int(0);
56 den_ = Int(1);
57 return;
58 }
59
60 // Move sign into numerator so that den_ > 0.
61 if (den_.sign_trit() == Trit::N) {
62 den_ = -den_;
63 num_ = -num_;
64 }
65
66 // 0/x → 0/1
67 if (num_.is_zero()) {
68 den_ = Int(1);
69 return;
70 }
71
72 Int g = gcd(abs_int(num_), den_);
73 if (!g.is_zero() && !(g == Int(1))) {
74 num_ /= g;
75 den_ /= g;
76 }

100 // Best-effort conversion from double using continued fractions.
101 static T81Fraction from_double(double x, int max_iterations = 64) noexcept {
102 if (x == 0.0) {
103 return T81Fraction{};
104 }
105
106 bool neg = x < 0.0;
107 x = std::fabs(x);
108
109 Int integer(static_cast<std::int64_t>(x));
110 double frac = x - static_cast<double>(integer.to_int64());
111
112 Int p0(1), q0(0);
113 Int p1(integer), q1(1);
114
115 int iter = 0;
116 while (frac > 0.0 && iter++ < max_iterations) {
117 const double r = 1.0 / frac;
118 const std::int64_t a = static_cast<std::int64_t>(r);
119
120 Int next_p = Int(a) * p1 + p0;
121 Int next_q = Int(a) * q1 + q0;
122
123 p0 = p1; q0 = q1;
124 p1 = next_p; q1 = next_q;
125
126 frac = r - static_cast<double>(a);
127 }
128
129 T81Fraction f(neg ? -p1 : p1, q1);
130 return f;
131 }
