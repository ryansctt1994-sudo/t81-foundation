34 inline const std::array<int16_t, 256>& get_byte_to_ternary() {
35 static const auto table = []() {
36 std::array<int16_t, 256> t{};
37 for (int i = 0; i < 256; ++i) {
38 int val = 0;
39 int p3 = 1;
40 for (int j = 0; j < 4; ++j) {
41 int u = (i >> (j * 2)) & 0x3;
42 if (u > 2) u = 1; // Treat invalid as Zero
43 val += (u - 1) * p3;
44 p3 *= 3;
45 }
46 t[i] = static_cast<int16_t>(val);
47 }
48 return t;
49 }();
50 return table;
51 }

53 inline const std::array<uint8_t, 81>& get_ternary_to_packed() {
54 static const auto table = []() {
55 std::array<uint8_t, 81> t{};
56 for (int i = 0; i < 81; ++i) {
57 int val = 0;
58 int tmp = i;
59 for (int j = 0; j < 4; ++j) {
60 int d = tmp % 3;
61 tmp /= 3;
62 val |= (d << (j * 2));
63 }
64 t[i] = static_cast<uint8_t>(val);
65 }
66 return t;
67 }();
68 return table;
69 }

104 void assign_from_int64(std::int64_t v) {
105 limbs_.clear();
106 if (v < 0) {
107 negative_ = true;
108 // std::int64_t min value is safe: abs(min) fits in 81 trits.
109 // Use unsigned to avoid overflow warning/UB when negating INT64_MIN.
110 const std::uint64_t uv = (v == std::numeric_limits<std::int64_t>::min())
111 ? static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1
112 : static_cast<std::uint64_t>(-v);
113 limbs_.emplace_back(static_cast<std::int64_t>(uv));
114 } else {
115 negative_ = false;
116 limbs_.emplace_back(v);
117 }
118 normalize();
119 }

159 [[nodiscard]] std::int64_t to_int64() const {
160 if (limbs_.empty()) {
161 throw std::logic_error("T81BigInt::to_int64: no limbs");
162 }
163 if (limbs_.size() > 1) {
164 throw std::overflow_error("T81BigInt::to_int64: value too large");
165 }
166
167 const std::int64_t mag = limbs_[0].to_int64(); // magnitude
168 if (mag < 0) {
169 // This can happen if the limb was constructed from a negative int64
170 // but we want magnitude. Actually assign_from_int64 handles this.
171 if (mag == std::numeric_limits<std::int64_t>::min()) {
172 // abs(min) still fits in 81 trits but not in int64_t magnitude.
173 throw std::overflow_error("T81BigInt::to_int64: magnitude overflow");
174 }
175 return negative_ ? mag : -mag; // should not reach here if normalize is correct
176 }
177
178 if (!negative_) {
179 return mag;
180 }
181
182 return -mag;
183 }

203 // Balanced-ternary string representation.
204 // Digits: '-', '0', '+'
205 [[nodiscard]] std::string str() const {
206 if (is_zero()) {
207 return "0";
208 }
209
210 std::string s;
211 for (size_t i = 0; i < limbs_.size(); ++i) {
212 for (size_t t = 0; t < kLimbTrits; ++t) {
213 Trit tr = limbs_[i][t];
214 // Magnitude trits are always non-negative in our canonical form
215 // after addition/subtraction adjustment.
216 if (tr == Trit::P) s.push_back('+');
217 else if (tr == Trit::N) s.push_back('-');
218 else s.push_back('0');
219 }
220 }
221
222 // Trim leading zeros (at the end of the string before reverse)
223 while (s.size() > 1 && s.back() == '0') s.pop_back();
224
225 if (negative_) {
226 s.push_back('-');
227 }
228
229 std::reverse(s.begin(), s.end());
230 return s;
231 }

293 1LL, 81LL, 6561LL, 531441LL, 43046721LL, 3486784401LL, 282429536481LL

315 if (x.negative_) c0 = -c0;

354 friend T81BigInt operator+(const T81BigInt& a, const T81BigInt& b) {
355 if (a.is_zero()) return b;
356 if (b.is_zero()) return a;
357 auto ac = get_chunks_static(a);
358 auto bc = get_chunks_static(b);
359 size_t n = std::max(ac.size(), bc.size());
360 size_t n_padded = (n + 3) & ~size_t(3);
361 ac.resize(n_padded, 0);
362 bc.resize(n_padded, 0);
363 static constexpr int64_t B = 7625597484987LL;
364 static constexpr int64_t halfB = (B - 1) / 2;
365 std::vector<int64_t> rc(n_padded + 1, 0);
366 size_t i = 0;
367#if defined(__AVX2__)
368 for (; i < n_padded; i += 4) {
369 __m256i va = _mm256_loadu_si256((const __m256i*)&ac[i]);
370 __m256i vb = _mm256_loadu_si256((const __m256i*)&bc[i]);
371 __m256i vsum = _mm256_add_epi64(va, vb);
372 _mm256_storeu_si256((__m256i*)&rc[i], vsum);
373 }
374#endif
375 for (; i < n; ++i) rc[i] = ac[i] + bc[i];
376 int64_t carry = 0;
377 for (size_t j = 0; j < rc.size() || carry != 0; ++j) {
378 if (j >= rc.size()) rc.push_back(0);
379 int64_t sum = rc[j] + carry;
380 if (sum > halfB) { rc[j] = sum - B; carry = 1; }
381 else if (sum < -halfB) { rc[j] = sum + B; carry = -1; }
382 else { rc[j] = sum; carry = 0; }
383 }
384 T81BigInt res;
385 res.limbs_.clear();
386 int64_t last = 0;
387 for (size_t i = rc.size(); i-- > 0; ) if (rc[i] != 0) { last = rc[i]; break; }
388 if (last < 0) {
389 res.negative_ = true;
390 int64_t c_neg = 0;
391 for (auto& v : rc) {
392 int64_t val = -v + c_neg;
393 if (val > halfB) { v = val - B; c_neg = 1; }
394 else if (val < -halfB) { v = val + B; c_neg = -1; }
395 else { v = val; c_neg = 0; }
396 }
397 } else res.negative_ = false;
398 const auto& packed_table = detail::get_ternary_to_packed();
399 for (size_t i = 0; i < rc.size(); i += 3) {
400 Limb l;
401 auto& ldata = const_cast<std::array<uint8_t, Limb::kNumBytes>&>(l.raw_data());
402 std::fill(ldata.begin(), ldata.end(), 0x55u);
403 auto set_chunk_in_limb = [&](int c, int64_t v) {
404 static constexpr int64_t offset27 = 3812798742493LL;
405 uint64_t uv = static_cast<uint64_t>(v + offset27);
406 int start_trit = c * 27;
407 for (int j = 0; j < 6; ++j) {
408 int r = uv % 81; uv /= 81;
409 uint8_t packed = packed_table[r];
410 for (int t = 0; t < 4; ++t) {
411 int u = (packed >> (t * 2)) & 0x3;
412 l[start_trit + j * 4 + t] = static_cast<Trit>(u - 1);
413 }
414 }
415 for(int t=24; t<27; ++t) { int r = uv % 3; uv /= 3; l[start_trit + t] = static_cast<Trit>(r - 1); }
416 };
417 for (int c = 0; c < 3; ++c) {
418 int64_t v = (i + c < rc.size()) ? rc[i + c] : 0;
419 if (res.negative_) v = -v;
420 set_chunk_in_limb(c, v);
421 }
422 res.limbs_.push_back(l);
423 }
424 res.normalize();
425 return res;
426 }

442 static std::vector<int128_t> karatsuba_mul_(const std::vector<int64_t>& a, const std::vector<int64_t>& b) {
443 size_t n = std::max(a.size(), b.size());
444 if (n <= 12) { // Schoolbook threshold
445 std::vector<int128_t> res(a.size() + b.size(), 0);
446 for (size_t i = 0; i < a.size(); ++i) {
447 const int64_t val_a = a[i];
448 if (val_a == 0) continue;
449 const int128_t a128 = val_a;
450 for (size_t j = 0; j < b.size(); ++j) {
451 const int64_t val_b = b[j];
452 if (val_b == 0) continue;
453 res[i + j] += a128 * val_b;
454 }
455 }
456 return res;
457 }
458
459 size_t k = n / 2;
460 auto split = [k](const std::vector<int64_t>& v) {
461 std::vector<int64_t> low, high;
462 for (size_t i = 0; i < v.size(); ++i) {
463 if (i < k) low.push_back(v[i]);
464 else high.push_back(v[i]);
465 }
466 if (low.empty()) low.push_back(0);
467 if (high.empty()) high.push_back(0);
468 return std::make_pair(low, high);
469 };
470
471 auto [a0, a1] = split(a);
472 auto [b0, b1] = split(b);
473
474 auto z0 = karatsuba_mul_(a0, b0);
475 auto z2 = karatsuba_mul_(a1, b1);
476
477 auto add_v = [](const std::vector<int64_t>& x, const std::vector<int64_t>& y) {
478 size_t nx = x.size(), ny = y.size();
479 size_t n = std::max(nx, ny);
480 std::vector<int64_t> r(n, 0);
481 size_t i = 0;
482#if defined(__AVX2__)
483 size_t n_simd = std::min(nx, ny) & ~size_t(3);
484 for (; i < n_simd; i += 4) {
485 __m256i vx = _mm256_loadu_si256((const __m256i*)&x[i]);
486 __m256i vy = _mm256_loadu_si256((const __m256i*)&y[i]);
487 _mm256_storeu_si256((__m256i*)&r[i], _mm256_add_epi64(vx, vy));
488 }
489#endif
490 for (; i < n; ++i) {
491 r[i] = (i < nx ? x[i] : 0) + (i < ny ? y[i] : 0);
492 }
493
494 int128_t carry = 0;
495 const int128_t B = 7625597484987LL;
496 const int128_t halfB = (B - 1) / 2;
497 for (size_t j = 0; j < r.size() || carry != 0; ++j) {
498 if (j >= r.size()) r.push_back(0);
499 int128_t val = r[j] + carry;
500 int128_t q = (val >= 0) ? (val + halfB) / B : (val - halfB) / B;
501 r[j] = static_cast<int64_t>(val - q * B);
502 carry = q;
503 }
504 return r;
505 };
506
507 auto a01 = add_v(a0, a1);
508 auto b01 = add_v(b0, b1);
509 auto z1 = karatsuba_mul_(a01, b01);
510
511 // z1 = z1 - z0 - z2
512 auto sub_v_128 = [](std::vector<int128_t>& x, const std::vector<int128_t>& y) {
513 size_t nx = x.size(), ny = y.size();
514 if (nx < ny) x.resize(ny, 0);
515 for (size_t i = 0; i < ny; ++i) x[i] -= y[i];
516 };
517
518 std::vector<int128_t> middle = z1;
519 sub_v_128(middle, z0);
520 sub_v_128(middle, z2);
521
522 std::vector<int128_t> res(z2.size() + 2 * k, 0);
523 if (res.size() < z0.size()) res.resize(z0.size(), 0);
524 if (res.size() < middle.size() + k) res.resize(middle.size() + k, 0);
525
526 for (size_t i = 0; i < z0.size(); ++i) res[i] += z0[i];
527 for (size_t i = 0; i < middle.size(); ++i) res[i + k] += middle[i];
528 for (size_t i = 0; i < z2.size(); ++i) res[i + 2 * k] += z2[i];
529
530 return res;
531 }

533 friend T81BigInt operator*(const T81BigInt& a, const T81BigInt& b) {
534 if (a.is_zero() || b.is_zero()) return T81BigInt::zero();
535 auto ac = get_chunks_static(a);
536 auto bc = get_chunks_static(b);
537 std::vector<int128_t> rc = karatsuba_mul_(ac, bc);
538
539 const int128_t B = 7625597484987LL; // 3^27
540 const int128_t halfB = (B - 1) / 2;
541 int128_t carry = 0;
542 std::vector<int64_t> final_c;
543 for (size_t i = 0; i < rc.size() || carry != 0; ++i) {
544 int128_t val = (i < rc.size() ? rc[i] : 0) + carry;
545 int128_t q = (val >= 0) ? (val + halfB) / B : (val - halfB) / B;
546 final_c.push_back(static_cast<int64_t>(val - q * B));
547 carry = q;
548 }
549
550 T81BigInt res;
551 res.limbs_.clear();
552 for (size_t i = 0; i < final_c.size(); i += 3) {
553 Limb l;
554 for (int c = 0; c < 3; ++c) {
555 int64_t v = (i + c < final_c.size()) ? final_c[i + c] : 0;
556 bool v_neg = v < 0;
557 uint64_t uv = v_neg ? static_cast<uint64_t>(-v) : static_cast<uint64_t>(v);
558 for (int t = 0; t < 27; ++t) {
559 int r = static_cast<int>(uv % 3); uv /= 3;
560 if (r == 2) { r = -1; uv++; }
561 l[c * 27 + t] = int_to_trit(v_neg ? -r : r);
562 }
563 }
564 res.limbs_.push_back(l);
565 }
566
567 res.negative_ = (a.negative_ != b.negative_);
568
569 // Correct sign if magnitude ended up negative (can happen due to balanced ternary representation)
570 Trit s = Trit::Z;
571 for (size_t i = res.limbs_.size(); i-- > 0; ) {
572 s = res.limbs_[i].sign_trit();
573 if (s != Trit::Z) break;
574 }
575 if (s == Trit::N) {
576 res.negative_ = !res.negative_;
577 int c_neg = 0;
578 for (auto& limb : res.limbs_) {
579 for (size_t t = 0; t < kLimbTrits; ++t) {
580 int v = -trit_to_int(limb[t]) + c_neg;
581 int d = (v > 1) ? v - 3 : (v < -1) ? v + 3 : v;
582 c_neg = (v > 1) ? 1 : (v < -1) ? -1 : 0;
583 limb[t] = int_to_trit(d);
584 }
585 }
586 }
587
588 res.normalize();
589 return res;
590 }
