81 constexpr Trit get_trit(size_type idx) const noexcept {
82 const size_type byte = idx / kTritsPerByte;
83 const size_type off = (idx % kTritsPerByte) * 2;
84 return decode_trit(static_cast<std::uint8_t>((data_[byte] >> off) & 0x3u));

87 constexpr void set_trit(size_type idx, Trit t) noexcept {
88 const size_type byte = idx / kTritsPerByte;
89 const size_type off = (idx % kTritsPerByte) * 2;
90 const std::uint8_t mask = static_cast<std::uint8_t>(~(0x3u << off));
91 const std::uint8_t enc = static_cast<std::uint8_t>(encode_trit(t) << off);
92 data_[byte] = (data_[byte] & mask) | enc;
93 }

154 void assign_from_int64(std::int64_t v) {
155 clear();
156 if (v == 0) return;
157 bool neg = v < 0;
158 std::uint64_t uv = (v == std::numeric_limits<std::int64_t>::min())
159 ? static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1
160 : static_cast<std::uint64_t>(neg ? -v : v);
161 size_type i = 0;
162 while (uv != 0 && i < kNumTrits) {
163 int r = static_cast<int>(uv % 3);
164 uv /= 3;
165 if (r == 2) { r = -1; ++uv; }
166 set_trit(i++, int_to_trit(neg ? -r : r));
167 }
168 if (uv != 0) throw std::overflow_error("T81Int: value does not fit in N trits");
169 }

172 // Universal C++20 to_int64(): works on MSVC (no throw), GCC/Clang (full safety)
173 [[nodiscard]] constexpr std::int64_t to_int64() const {
174#ifdef _MSC_VER
175 // MSVC: no throw allowed in constexpr → fast path (symbol IDs are tiny)
176 std::int64_t value = 0;
177 std::int64_t pow3 = 1;
178 const size_type limit = (kNumTrits < kMaxSafeTrits) ? kNumTrits : kMaxSafeTrits;
179 for (size_type i = 0; i < limit; ++i) {
180 value += trit_to_int(get_trit(i)) * pow3;
181 if (i < kMaxSafeTrits - 1) pow3 *= 3;
182 }
183 return value;
184#else
185 // GCC/Clang: full overflow checking with throw (safe in constexpr if not executed)
186 const size_type limit = (kNumTrits < kMaxSafeTrits) ? kNumTrits : kMaxSafeTrits;
187
188 for (size_type i = limit; i < kNumTrits; ++i) {
189 if (get_trit(i) != Trit::Z)
190 throw std::overflow_error("T81Int::to_int64(): value out of range");
191 }
192
193 std::int64_t value = 0;
194 std::int64_t pow3 = 1;
195 for (size_type i = 0; i < limit; ++i) {
196 const int d = trit_to_int(get_trit(i));
197 if (d != 0) {
198 const std::int64_t term = d * pow3;
199 if (term > 0 && value > std::numeric_limits<std::int64_t>::max() - term)
200 throw std::overflow_error("T81Int::to_int64(): overflow (positive)");
201 if (term < 0 && value < std::numeric_limits<std::int64_t>::min() - term)
202 throw std::overflow_error("T81Int::to_int64(): overflow (negative)");
203 value += term;
204 }
205 if (i + 1 < limit) pow3 *= 3;
206 }
207 return value;
208#endif

265 friend constexpr T81Int operator+(const T81Int& a, const T81Int& b) {
266 T81Int r;
267 int carry = 0;
268 for (size_type i = 0; i < kNumTrits; ++i) {
269 int sum = trit_to_int(a.get_trit(i)) + trit_to_int(b.get_trit(i)) + carry;
270 int digit = (sum > 1) ? sum - 3 : (sum < -1) ? sum + 3 : sum;
271 carry = (sum > 1) ? 1 : (sum < -1) ? -1 : 0;
272 r.set_trit(i, int_to_trit(digit));
273 }
274 if (carry != 0) {

276 }

311 std::string to_string() const {
312 std::int64_t v = to_int64();
313 if (v == 0) return "0";
314 bool neg = v < 0;
315 if (neg) v = -v;
316 std::string s;
317 while (v) {
318 int r = static_cast<int>(v % 3);
319 v /= 3;
320 s.push_back(r == 0 ? '0' : r == 1 ? '1' : '2');
321 }
322 if (neg) s.push_back('-');
323 std::reverse(s.begin(), s.end());
324 return s;
325 }
