112 // ================================================================
113 // Slice – safe subrange extraction
114 // ================================================================
115 //
116 // Returns a new T81Bytes containing at most `length` bytes starting
117 // at `offset`. If offset >= size, returns an empty T81Bytes.
118 [[nodiscard]] T81Bytes slice(size_type offset, size_type length) const {
119 if (offset >= data_.size()) {
120 return T81Bytes{};
121 }
122 const size_type end_index =
123 std::min<size_type>(data_.size(), offset + length);
124 container_type sub;
125 sub.reserve(end_index - offset);
126 sub.insert(
127 sub.end(),
128 data_.begin() + static_cast<std::ptrdiff_t>(offset),
129 data_.begin() + static_cast<std::ptrdiff_t>(end_index)
130 );
131 return T81Bytes(std::move(sub));
132 }
