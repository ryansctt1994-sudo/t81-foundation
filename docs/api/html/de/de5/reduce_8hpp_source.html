11// axis == 1 → reduce over cols (per-row result) → shape {R}

15 const auto& d = m.data();

18 std::vector<float> out((size_t)C, 0.0f);
19 for (int r = 0; r < R; ++r) {
20 const size_t base = (size_t)r * C;
21 for (int c = 0; c < C; ++c) out[(size_t)c] += d[base + (size_t)c];
22 }

24 } else if (axis == 1) {
